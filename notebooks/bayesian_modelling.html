2 Define Priors and Configuration

We use biological bounds to prevent the frequentist singularities (D -> infinity, c -> 0).

# Common priors for the three Yield models
bprior_yield <- c(
  prior(normal(50, 50), nlpar = "Y0", lb = 0),     
  prior(normal(100, 50), nlpar = "A", lb = 0),     
  prior(lognormal(-2, 2), nlpar = "cbase", lb = 0), 
  prior(normal(0, 1), nlpar = "betainvb"),
  prior(normal(0, 1), nlpar = "betaN"),
  prior(normal(0, 1), nlpar = "betaTemp"),
  prior(normal(0, 1), nlpar = "betaPrec")
)

# Priors for the Uptake model (Michaelis-Menten)
bprior_uptake <- c(
  prior(normal(30, 15), nlpar = "Vmax", lb = 0),
  prior(lognormal(1, 1), nlpar = "Kbase", lb = 0),
  prior(normal(0, 1), nlpar = "betainvb"),
  prior(normal(0, 1), nlpar = "betaN"),
  prior(normal(0, 1), nlpar = "betaTemp"),
  prior(normal(0, 1), nlpar = "betaPrec")
)

# Priors for the CP model (Mitscherlich)
bprior_cp <- c(
  prior(normal(1.5, 1), nlpar = "Cmin", lb = 0),
  prior(normal(2.0, 1.5), nlpar = "DeltaC", lb = 0),
  prior(lognormal(-2, 2), nlpar = "cbase", lb = 0),
  prior(normal(0, 1), nlpar = "betainvb"),
  prior(normal(0, 1), nlpar = "betaTemp"),
  prior(normal(0, 1), nlpar = "betaPrec")
)

# HMC Sampling config
sampling_iter <- 2000
sampling_chains <- 4
sampling_cores <- 4

3 Model 1: Empirical Quantity (\(P_{\text{AAE10}}\))

bform_aae <- bf(
  total_yield ~ Y0 + (A - Y0) * (1 - exp(-(cbase * exp(betainvb * z_inv_b_cons + betaN * z_fert_N + betaTemp * z_Temp_Mean + betaPrec * z_Prec_Anom) * soil_0_20_P_AAE10))),
  Y0 ~ crop - 1 + (1 | site/year),
  A ~ crop - 1 + (1 | site/year),
  cbase ~ crop - 1,
  betainvb + betaN + betaTemp + betaPrec ~ 1,
  nl = TRUE
)

# Fit model (uses cmdstanr or rstan backend natively if configured, we just run brm)
m_brms_raw_aae <- brm(
  formula = bform_aae,
  data = d_brms,
  prior = bprior_yield,
  family = gaussian(),
  backend = "cmdstanr",
  threads = threading(2),
  chains = sampling_chains, 
  cores = sampling_cores, 
  iter = sampling_iter,
  control = list(adapt_delta = 0.95),
  seed = 42,
  file = "models/brms_yield_raw_aae"
)
