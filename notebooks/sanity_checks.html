4 9. Site-Dependency of Environmental Covariates (Variance Decomposition)

As observed above, metrics like the cumulative P balance and yield responses appear highly site-dependent. To formally quantify how strictly a covariate is grouped by its site (e.g. soil geology) versus how much it varies independently across plots and years (e.g. fertilizer, weather), we can compute the Intra-Class Correlation Coefficient (ICC).

4.1 The Mixed-Effects Variance Decomposition

To formally quantify how strictly a covariate is grouped by its site (e.g. soil geology) versus how much it varies independently across plots and years (e.g. fertilizer, weather), we compute the Intra-Class Correlation Coefficient (ICC).

For every numeric parameter in the dataset, we fit a random intercept mixed-effects model with no fixed effects (lmer(scale(variable) ~ 1 + (1 | site))). This forces the model to decompose the total variance of the parameter into exactly two variance fractions: 1. Between-Site Variance (\(\sigma^2_{site}\)): Variance driven purely by moving from one geographical location to another (geology and macro-climate). 2. Residual Variance (\(\sigma^2_{residual}\)): “Noise” happening within the site—such as plot-to-plot differences (caused by P, K, Mg fertilizer treatments) and year-to-year differences (weather anomalies, crop rotations).

The ICC is then calculated as the ratio of site-driven variance to total variance: \[ ICC = \frac{\sigma^2_{site}}{\sigma^2_{site} + \sigma^2_{residual}} \]

  • ICC > 0.80: The variable is almost entirely dictated by the geographical site (e.g. pH or soil texture). These parameters act as excellent, stable candidates for modeling the physical pedoclimatic baseline (the \(1/b\) physical highway).
  • ICC < 0.30: The variable fluctuates completely independently of the site. Its variance is heavily driven by local experimental treatments (e.g., fert_N, P_CO2) or local weather (Temp_Anom). Nesting these into the models (like the Mitscherlich rate constant \(c_{eff}\)) helps capture critical plot-level nutritional variance that the rigid site-level variables are blind to.
library(lme4)
library(performance)
library(purrr)

# Define the variables we want to decompose
# We select all numeric variables from D_main, excluding kinetics and identifiers
num_cols <- names(D_main)[sapply(D_main, is.numeric)]
covariates <- setdiff(num_cols, c(
    "year", "rep", "k", "Pmax_PS", "v0_kPS", "soil_0_20_P_test"
))

# Helper function to compute ICC for a single variable
get_icc <- function(var_name, data) {
    # We use a tryCatch just in case a variable is missing or singular
    res <- tryCatch({
        # Drop NAs for this specific variable to fit the lmer
        d_clean <- data |> filter(!is.na(.data[[var_name]]))
        
        # Build formula explicitly because lmer doesn't support the dplyr .data pronoun
        fmla <- as.formula(paste0("scale(`", var_name, "`) ~ 1 + (1 | site)"))
        mod <- lmer(fmla, data = d_clean, control = lmerControl(calc.derivs = FALSE))
        
        # Explicitly extract variance components
        vc <- as.data.frame(VarCorr(mod))
        var_site <- vc$vcov[vc$grp == "site"]
        var_resid <- vc$vcov[vc$grp == "Residual"]
        
        icc_val <- var_site / (var_site + var_resid)
        return(icc_val)
    }, error = function(e) return(NA))
    
    return(res)
}

# Calculate ICC for all covariates
icc_results <- data.frame(
    Variable = covariates,
    ICC = sapply(covariates, get_icc, data = D_main)
) |> 
    filter(!is.na(ICC)) |> # Drop variables that failed to compute
    arrange(desc(ICC)) |>
    mutate(
        Category = case_when(
            ICC > 0.8 ~ "Highly Site-Dependent (Geology/Climate Baseline)",
            ICC > 0.3 ~ "Mixed (Site + Plot/Temporal Effects)",
            TRUE ~ "Highly Local/Temporal (Treatments/Weather)"
        )
    )

4.2 10. Covariate Permutation: Identifying the Best Pedoclimatic Drivers

To rigorously test which combination of pedoclimatic traits best explains the variance across our 7 sites, we fit a series of Mixed-Effects models. For each of our 3 main soil indicators (\(P_{CO2}\), \(P_{AAE10}\), \(PS\)), we fit a baseline model: annual_P_balance ~ scale(Indicator) + (1 | site)

Then, we systematically loop through every 2-variable combination of our 6 pedoclimatic drivers (pH, Corg, Clay, Lime, Temp, Prec) and fit: annual_P_balance ~ scale(Indicator) + scale(Var1) + scale(Var2) + (1 | site)

If the two physical variables successfully explain the site-to-site differences, the Random Intercept variance (1 | site) should shrink substantially towards 0.

# Function to run permutations
run_perms <- function(target, inds, covs, dat) {
    d_clean <- dat |> drop_na(all_of(c(target, "site", covs, inds)))
    combos <- combn(covs, 2, simplify = FALSE)
    results <- list()
    for(ind in inds) {
        # Baseline model
        f_base <- as.formula(paste0("scale(", target, ") ~ scale(`", ind, "`) + (1 | site)"))
        mod_base <- lmer(f_base, data = d_clean, control = lmerControl(calc.derivs = FALSE))
        base_vc <- as.data.frame(VarCorr(mod_base))
        base_site_var <- base_vc$vcov[base_vc$grp == "site"]
        
        for(cb in combos) {
            v1 <- cb[1]; v2 <- cb[2]
            f_mod <- as.formula(paste0("scale(", target, ") ~ scale(`", ind, "`) + scale(`", v1, "`) + scale(`", v2, "`) + (1 | site)"))
            mod <- suppressWarnings(tryCatch({
                lmer(f_mod, data = d_clean, control = lmerControl(calc.derivs = FALSE))
            }, error = function(e) NULL))
            
            if(is.null(mod)) next
            
            vc <- as.data.frame(VarCorr(mod))
            site_var <- vc$vcov[vc$grp == "site"]
            
            # Safely calculate Marginal R2
            r2m <- suppressWarnings(tryCatch(as.numeric(performance::r2(mod)$R2_marginal), error=function(e) NA))
            
            results[[length(results)+1]] <- data.frame(
                Soil_Indicator = ind,
                Pedo_Var_1 = v1,
                Pedo_Var_2 = v2,
                R2_Marginal = r2m,
                AIC = AIC(mod),
                Site_Var_Reduction_Pct = (base_site_var - site_var) / base_site_var * 100
            )
        }
    }
    
    bind_rows(results) |> 
        mutate(Stability = ifelse(Site_Var_Reduction_Pct < 0, "Unstable (Collinear)", "Stable")) |>
        filter(Stability == "Stable") |>
        mutate(
            R2_Marginal = round(R2_Marginal, 3),
            Site_Var_Reduction_Pct = round(Site_Var_Reduction_Pct, 1),
            AIC = round(AIC, 1)
        ) |>
        arrange(desc(R2_Marginal)) |>
        select(-Stability)
}

5 11. Final Task Sanity Graphs: Yield and Uptake Distributions

To verify the overall distributions and responses of Yield and P Uptake across the sites, treatments, years, and specific fertilizers, we plot the requested sanity checks below.

library(ggplot2)
library(dplyr)
library(tidyr)

# Calculate cumulative P uptake
D_graphs <- D_main |>
    group_by(site, plot_nr) |>
    arrange(year) |>
    mutate(
        cumulative_P_up = cumsum(replace_na(annual_P_uptake, 0))
    ) |>
    ungroup() |>
    filter(!is.na(annual_yield_mp_DM))

# Helper to plot across the 3 metrics
plot_sanity <- function(metric_y, title_y) {
    # 1. Y ~ Treatment faceted by trial
    p1 <- ggplot(D_graphs, aes(x = Treatment, y = .data[[metric_y]], fill = Treatment)) +
        geom_boxplot(alpha = 0.7, outlier.alpha = 0.3) +
        facet_wrap(~ site, scales = "free_y") +
        theme_bw() + 
        theme(axis.text.x = element_text(angle = 45, hjust = 1)) +
        labs(title = paste(title_y, "~ Treatment by Site"), x = "Treatment", y = title_y)
    
    # 2. Y ~ year faceted by crop and site
    p2 <- ggplot(D_graphs, aes(x = year, y = .data[[metric_y]], color = crop)) +
        geom_point(alpha = 0.5) +
        facet_grid(crop ~ site, scales = "free_y") +
        theme_bw() +
        theme(strip.text.y = element_text(angle = 0), legend.position = "none", axis.text.x = element_text(angle = 45, hjust = 1)) +
        labs(title = paste(title_y, "~ Year faceted by Crop and Site"), x = "Year", y = title_y)
    
    # 3. Y ~ fert_X for X in {P, K, Mg, N}
    D_fert <- D_graphs |>
        select(site, Treatment, all_of(metric_y), fert_P_tot, fert_K_tot, fert_Mg_tot, fert_N_tot) |>
        pivot_longer(cols = starts_with("fert_"), names_to = "Fertilizer", values_to = "Amount") |>
        mutate(Fertilizer = factor(Fertilizer, levels = c("fert_P_tot", "fert_K_tot", "fert_Mg_tot", "fert_N_tot"),
                                   labels = c("P Applied", "K Applied", "Mg Applied", "N Applied")))
        
    p3 <- ggplot(D_fert, aes(x = Amount, y = .data[[metric_y]], color = site)) +
        geom_point(alpha = 0.3) +
        geom_smooth(method = "lm", se = FALSE) +
        facet_wrap(~ Fertilizer, scales = "free_x") +
        theme_bw() +
        labs(title = paste(title_y, "~ Fertilizer Applied"), x = "Fertilizer Amount (kg/ha)", y = title_y)
    
    print(p1)
    print(p2)
    print(p3)
}

plot_sanity("annual_yield_mp_DM", "Annual Dry Yield (kg/ha)")
plot_sanity("annual_P_uptake", "Annual P Uptake (kg P/ha)")
plot_sanity("cumulative_P_up", "Cumulative P Uptake (kg P/ha)")
